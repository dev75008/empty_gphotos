(async () => {
  const SUPPRIMER = false;   // false = test (sélectionne seulement). true = supprime vraiment.
  const LOT = 1000;          // nombre de photos par lot
  const PAUSE = 1500;
  const MAX_TOURS = 2000;
  const sleep = ms => new Promise(r => setTimeout(r, ms));

  if (!/photos\.google\.com\/?(u\/\d+\/?)?(\?.*)?$/.test(location.href)) {
    console.error('⛔ Ouvre la page principale "Photos" (pas la corbeille, un album, une recherche), puis relance.');
    return;
  }

  const libelle = el =>
    ((el.getAttribute('aria-label') || '') + ' ' +
     (el.getAttribute('title') || '') + ' ' +
     (el.getAttribute('data-tooltip') || '')).trim();

  const visible = el => {
    const r = el.getBoundingClientRect();
    const s = getComputedStyle(el);
    return r.width > 0 && r.height > 0 && r.bottom > 0 && r.top < innerHeight &&
           s.visibility !== 'hidden' && s.display !== 'none';
  };

  const survoler = el => {
    for (const t of ['pointerover', 'pointerenter', 'mouseover', 'mouseenter', 'mousemove']) {
      el.dispatchEvent(new MouseEvent(t, { bubbles: true }));
    }
  };

  let cache = null;
  const conteneur = () => {
    if (cache && cache.isConnected) return cache;
    let best = document.scrollingElement, max = 0;
    for (const el of document.querySelectorAll('*')) {
      const s = getComputedStyle(el);
      if (/(auto|scroll)/.test(s.overflowY) && el.scrollHeight > el.clientHeight + 50) {
        const surface = el.clientHeight * el.clientWidth;
        if (surface > max) { max = surface; best = el; }
      }
    }
    return (cache = best);
  };
  const defiler = () => {
    const c = conteneur();
    const avant = c.scrollTop;
    c.scrollBy(0, c.clientHeight * 0.8);
    return c.scrollTop > avant;
  };
  const remonter = () => conteneur().scrollTo(0, 0);

  const casesLibres = () =>
    [...document.querySelectorAll('[role="checkbox"][aria-checked="false"]')].filter(visible);

  const REGEX_SUPPR = /corbeille|supprimer|trash|delete/i;
  const REGEX_EXCLU = /souvenir|memor|album|définitivement|permanently|partag|shar/i;

  const boutonCorbeille = () =>
    [...document.querySelectorAll('button, [role="button"]')]
      .filter(visible)
      .filter(el => !el.closest('[role="dialog"], [role="alertdialog"], nav, [role="navigation"]'))
      .find(el => REGEX_SUPPR.test(libelle(el)) && !REGEX_EXCLU.test(libelle(el)));

  const boutonConfirmer = () =>
    [...document.querySelectorAll('[role="dialog"] button, [role="alertdialog"] button, [role="dialog"] [role="button"], [role="alertdialog"] [role="button"]')]
      .filter(visible)
      .find(b => {
        const t = b.textContent + ' ' + libelle(b);
        return REGEX_SUPPR.test(t) && !REGEX_EXCLU.test(t);
      });

  const echap = () => {
    const opts = { key: 'Escape', code: 'Escape', keyCode: 27, bubbles: true };
    (document.activeElement || document.body).dispatchEvent(new KeyboardEvent('keydown', opts));
  };

  let lot = LOT, lots = 0, total = 0, vides = 0;
  remonter();
  await sleep(PAUSE);

  for (let tour = 1; tour <= MAX_TOURS; tour++) {
    let n = 0;
    for (let s = 0; s < 6000 && n < lot; s++) {
      for (const c of casesLibres()) {
        survoler(c);
        c.click();
        n++;
        await sleep(30);
        if (n >= lot) break;
      }
      if (n >= lot) break;
      if (!defiler()) { await sleep(1200); if (!defiler()) break; }
      await sleep(600);
    }

    if (n === 0) {
      if (++vides >= 3) {
        console.log(`✅ Terminé. ${lots} lot(s), ${total} photo(s) envoyée(s) à la corbeille.`);
        break;
      }
      console.warn('⚠️ Aucune case trouvée sur ce passage, nouvel essai...');
      remonter(); await sleep(PAUSE * 2);
      continue;
    }
    vides = 0;
    console.log(`☑️ ${n} photo(s) cliquée(s).`);
    await sleep(PAUSE);

    if (!SUPPRIMER) {
      const b = boutonCorbeille();
      console.log('🛑 Mode test : rien n\'est supprimé. Regarde en haut à gauche combien de photos sont sélectionnées.');
      console.log(b ? `Bouton corbeille trouvé : "${libelle(b)}"` : 'Bouton corbeille NON trouvé.');
      if (!b) {
        console.log('Boutons visibles :\n' + [...document.querySelectorAll('button, [role="button"]')]
          .filter(visible).map(libelle).filter(Boolean).join('\n'));
      }
      return;
    }

    remonter(); await sleep(PAUSE);
    const corbeille = boutonCorbeille();
    if (corbeille) corbeille.click();
    let confirmer = null;
    for (let i = 0; i < 10 && !confirmer; i++) { await sleep(PAUSE); confirmer = boutonConfirmer(); }

    if (!confirmer) {
      echap(); await sleep(PAUSE);
      lot = Math.floor(lot / 2);
      if (lot < 10) {
        console.warn('❌ Suppression impossible. Recharge la page (F5) et réessaie.');
        break;
      }
      console.warn(`⚠️ Suppression échouée : nouvel essai avec des lots de ${lot}. Recharge la page (F5) si des photos restent cochées.`);
      remonter(); await sleep(PAUSE * 2);
      continue;
    }
    confirmer.click();
    lots++; total += n;
    console.log(`🗑️ Lot ${lots} supprimé (${n} photo(s), total ${total})`);
    await sleep(PAUSE * 4);
    remonter();
    await sleep(PAUSE * 2);
  }
})();
